\documentclass[tikz,border=8pt]{standalone}
\usepackage{ctex}
\usepackage{tikz-uml}
\usetikzlibrary{arrows.meta,positioning}

\begin{document}
\begin{tikzpicture}[
    font=\sffamily\small,
    umltext/.style={font=\sffamily\small}
]

% ===== Actors outside the system boundary =====
\umlactor[name=user,y=2]{用户\\(User)}
\umlactor[name=dba,y=-4.5]{数据库管理员\\(DBA)}

% ===== System boundary and use cases =====
\begin{umlsystem}[x=4,fill=blue!2]{支持混合隔离级别的分布式数据库}
    % --- 事务控制 ---
    \umlusecase[name=begin,    x=1.9, y=3.2,  fill=purple!10]{开启事务\\(Begin)}
    \umlusecase[name=isolation,x=6.2, y=3.2,  fill=purple!10]{选择隔离\\级别}

    % --- 数据操作 ---
    \umlusecase[name=read,     x=1.9, y=1.5,  fill=cyan!12]{读取\\(Read)}
    \umlusecase[name=write,    x=1.9, y=0.0,  fill=cyan!12]{写入\\(Write)}
    \umlusecase[name=delete,   x=1.9, y=-1.5, fill=cyan!12]{删除\\(Delete)}

    % --- 提交流程 ---
    \umlusecase[name=validate, x=6.2, y=0.6,  fill=orange!18]{校验事务\\状态}
    \umlusecase[name=commit,   x=6.2, y=-1.0, fill=purple!10]{提交\\(Commit)}
    \umlusecase[name=abort,    x=6.2, y=-2.5, fill=purple!10]{中止\\(Abort)}

    % --- 共享：集群可观测 ---
    \umlusecase[name=status,   x=4.05,y=-4.0, fill=teal!18]{查看\\集群状态}

    % --- 集群运维（DBA） ---
    \umlusecase[name=conn,     x=1.9, y=-5.7, fill=green!14]{连接\\持续时间}
    \umlusecase[name=timeout,  x=6.2, y=-5.7, fill=green!14]{最长事务\\时间}
    \umlusecase[name=lb,       x=1.9, y=-7.2, fill=green!14]{负载均衡\\策略}
    \umlusecase[name=retry,    x=6.2, y=-7.2, fill=green!14]{重试策略}
\end{umlsystem}

% ===== 用户 -> 用例 =====
\umlassoc{user}{begin}
\umlassoc{user}{read}
\umlassoc{user}{write}
\umlassoc{user}{delete}
\umlassoc{user}{status}

% ===== DBA -> 用例 =====
\umlassoc{dba}{conn}
\umlassoc{dba}{lb}
\umlassoc{dba}{timeout}
\umlassoc{dba}{retry}
\umlassoc{dba}{status}

% ===== <<include>> 依赖 =====
\umlinclude{begin}{isolation}
\umlinclude{commit}{validate}
\umlinclude{abort}{validate}
\umlinclude{read}{validate}
\umlinclude{write}{validate}
\umlinclude{delete}{validate}

\end{tikzpicture}
\end{document}
