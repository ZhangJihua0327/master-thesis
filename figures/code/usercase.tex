\documentclass[tikz,border=10pt]{standalone}
\usepackage{ctex}
\usepackage{tikz-uml}
\usetikzlibrary{arrows.meta,positioning,fit}

\begin{document}
\begin{tikzpicture}[
    font=\sffamily\small,
    umltext/.style={font=\sffamily\small}
]

% ===== Actors outside the system boundary =====
\umlactor[name=user,y=2]{用户\\(User)}
\umlactor[name=dba,y=-4.8]{数据库管理员\\(DBA)}

% ===== System boundary and use cases =====
\begin{umlsystem}[x=4,fill=blue!2]{支持混合隔离级别的分布式数据库}
    % 透明 spacer：顶部撑出系统标题的空间，右边撑出 box 宽度
    \node[opacity=0,inner sep=0] (top-pad)   at (4.7, 4.3)  {.};
    \node[opacity=0,inner sep=0] (right-pad) at (10.0, 0)   {.};

    % --- 事务控制 ---
    \umlusecase[name=begin,    x=1.9, y=3.3,  fill=purple!10]{开启事务\\(Begin)}
    \umlusecase[name=isolation,x=7.6, y=3.3,  fill=purple!10]{选择隔离\\级别}

    % --- 数据操作（左列） ---
    \umlusecase[name=read,     x=1.9, y=1.3,  fill=cyan!12]{读取\\(Read)}
    \umlusecase[name=write,    x=1.9, y=-0.2, fill=cyan!12]{写入\\(Write)}
    \umlusecase[name=delete,   x=1.9, y=-1.7, fill=cyan!12]{删除\\(Delete)}

    % --- 校验 + 提交流程（右列） ---
    \umlusecase[name=commit,   x=7.6, y=1.5,  fill=purple!10]{提交\\(Commit)}
    \umlusecase[name=validate, x=7.6, y=-0.3, fill=orange!18]{校验事务\\状态}
    \umlusecase[name=abort,    x=7.6, y=-2.4, fill=purple!10]{中止\\(Abort)}

    % --- 共享：集群可观测（夹在数据操作与集群运维之间） ---
    \umlusecase[name=status,   x=4.75,y=-3.8, fill=teal!18]{查看\\集群状态}

    % --- 集群运维 2x2（DBA） ---
    \umlusecase[name=conn,     x=1.9, y=-5.2, fill=green!14]{连接\\持续时间}
    \umlusecase[name=timeout,  x=7.6, y=-5.2, fill=green!14]{最长事务\\时间}
    \umlusecase[name=lb,       x=1.9, y=-6.7, fill=green!14]{负载均衡\\策略}
    \umlusecase[name=retry,    x=7.6, y=-6.7, fill=green!14]{重试策略}
\end{umlsystem}

% ===== 用户 -> 用例（线条起点固定在小人右臂端点 (+0.35,+0.02)） =====
\draw[thick] ($(user)+(0.35,0.02)$) -- (begin.west);
\draw[thick] ($(user)+(0.35,0.02)$) -- (read.west);
\draw[thick] ($(user)+(0.35,0.02)$) -- (write.west);
\draw[thick] ($(user)+(0.35,0.02)$) -- (delete.west);
\draw[thick] ($(user)+(0.35,0.02)$) -- (status.west);
% user → commit / abort：弧形绕过数据操作区，避免直线穿 read/write/delete
\draw[thick] ($(user)+(0.35,0.02)$) to[bend left=20] (commit.west);
\draw[thick] ($(user)+(0.35,0.02)$) to[bend left=50] (abort.west);

% ===== DBA -> 用例（线条起点固定在小人右臂端点） =====
\draw[thick] ($(dba)+(0.35,0.02)$) -- (conn.west);
\draw[thick] ($(dba)+(0.35,0.02)$) -- (lb.west);
\draw[thick] ($(dba)+(0.35,0.02)$) -- (timeout.west);
\draw[thick] ($(dba)+(0.35,0.02)$) -- (retry.west);
% DBA → status：先上后右走 elbow
\draw[thick] ($(dba)+(0.35,0.02)$) |- (status.west);

% ===== <<include>> 依赖 =====
% begin→isolation 水平：起点指 begin.east、终点指 isolation.west，
% 不让线穿过椭圆中心
\umlinclude[arg1=east,arg2=west]{begin}{isolation}

% commit/abort→validate 用裸 \draw 绕到 validate 东侧，label 不 sloped
% 避免被 \umlinclude 内置的 sloped 旋转后压在 validate 椭圆上
\draw[dashed,thick,-{Latex[length=2mm]}] (commit.east) to[bend left=20]
    node[midway,right=1pt,font=\scriptsize]
    {\textless\textless include\textgreater\textgreater}
    (validate.east);
\draw[dashed,thick,-{Latex[length=2mm]}] (abort.east) to[bend right=20]
    node[midway,right=1pt,font=\scriptsize]
    {\textless\textless include\textgreater\textgreater}
    (validate.east);

\end{tikzpicture}
\end{document}
