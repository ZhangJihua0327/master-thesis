  \IfNoValueTF{#1}{%
    \IfNoValueTF{#2}{$\operatorname{read}()$}{$\operatorname{read}(#2)$}%
  }{%
    \IfNoValueTF{#2}{$\operatorname{read}(#1)$}{$\operatorname{read}(#1,#2)$}%
  }%
}

  }%

% 算法命令定义
\algrenewcommand\textproc{}
\renewcommand{\algorithmicprocedure}{\textbf{procedure}}
\algrenewcommand{\algorithmiccomment}[1]{\hfill $\triangleright$ #1}

% 重定义 \Procedure 和 \EndProcedure 以增加 Procedure 之间的间距
% 第一个参数自动使用 \textsc{} 包裹
\let\OriginalProcedure\Procedure
\let\OriginalEndProcedure\EndProcedure
\renewcommand{\Procedure}[2]{\vspace{0.2em}\OriginalProcedure{\textsc{#1}}{#2}}
\renewcommand{\EndProcedure}{\OriginalEndProcedure\vspace{0.2em}}

\let\OriginalFunction\Function
\let\OriginalEndFunction\EndFunction
\renewcommand{\Function}[2]{\vspace{0.2em}\OriginalFunction{\textsc{#1}}{#2}}
\renewcommand{\EndFunction}{\OriginalEndFunction\vspace{0.2em}}
